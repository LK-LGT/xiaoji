### 1.2 `backend/app/services/agent_tool_token.py` — 新建

完整内容（可直接粘贴）：

```python
"""智能体工具层短时令牌 —— 签发与校验。

设计要点（对齐 docs/01 四条原则 + 工具层五条边界）：
1. 令牌里**只有** employee_id 是身份来源，工具参数永远不接受 employee_id。
2. 白名单 `allow` 绑定具体工具，超出即 403。
3. 绑定 task，5 分钟过期（硬上限 300s）。
4. 签名算法与 routes/auth.py::_generate_token 一致，但**密钥独立**。

⚠️ 为什么不用 JWT：
   agent-service 的 requirements 只有 fastapi/uvicorn/pydantic，
   多一个 pyjwt 依赖就多一个"部署时忘了装"的坑。
   本仓库 auth 本来就是自签的 base64url(payload).hmac 格式，
   沿用同一套编码风格反而更一致、更好审计。
"""

from __future__ import annotations

import base64
import hashlib
import hmac
import json
import time
from collections.abc import Sequence
from dataclasses import dataclass

from ..core.config import settings

# TTL 硬上限 —— 即使调用方传了更大的值也会被钳制。
MAX_TTL_SECONDS = 300
MIN_TTL_SECONDS = 30


class AgentToolTokenError(Exception):
    """工具令牌错误基类。"""

    status_code = 401
    message = "工具令牌无效"


class AgentToolTokenExpired(AgentToolTokenError):
    status_code = 403
    message = "工具令牌已过期"


class AgentToolTokenForbidden(AgentToolTokenError):
    """令牌有效，但无权做这件事（如工具不在白名单）。"""

    status_code = 403
    message = "无权调用该工具"


@dataclass(frozen=True)
class AgentToolTokenPayload:
    """工具令牌载荷。

    ⚠️ `sub` 是**唯一**的身份来源。L2 工具网关从令牌取 employee_id，
       从不从请求体取 —— 这是防止 agent 查别人的第一道闸。
    """

    sub: str          # employee_id
    task: str         # 绑定的考核任务
    allow: tuple[str, ...]   # 工具白名单
    scope: str        # self | department（粗粒度，L2 侧仍会用 access_control 复核）
    iat: int
    exp: int


def _b64encode(raw: bytes) -> str:
    return base64.urlsafe_b64encode(raw).decode().rstrip("=")


def _b64decode(text: str) -> bytes:
    padding = "=" * (-len(text) % 4)
    return base64.urlsafe_b64decode(text + padding)


def _sign(encoded_payload: str) -> str:
    secret = settings.agent_tool_secret.encode("utf-8")
    return hmac.new(secret, encoded_payload.encode("utf-8"), hashlib.sha256).hexdigest()


def issue_agent_tool_token(
    *,
    employee_id: str,
    task_id: str,
    allow: Sequence[str],
    scope: str = "self",
    ttl: int | None = None,
) -> str:
    """签发短时工具令牌。

    ⚠️ **绝不通过 HTTP 暴露签发接口** —— 只有 L2 内部调用。
    """
    if not employee_id:
        raise ValueError("employee_id 不能为空")

    effective_ttl = ttl if ttl is not None else settings.agent_tool_token_ttl_seconds
    effective_ttl = max(MIN_TTL_SECONDS, min(MAX_TTL_SECONDS, int(effective_ttl)))

    now = int(time.time())
    payload = {
        "sub": employee_id,
        "task": task_id,
        # 排序去重 —— 保证同一组白名单签出同一个签名，便于比对与排障
        "allow": sorted(set(allow)),
        "scope": scope,
        "iat": now,
        "exp": now + effective_ttl,
    }
    encoded = _b64encode(
        json.dumps(payload, separators=(",", ":"), sort_keys=True).encode("utf-8")
    )
    return f"{encoded}.{_sign(encoded)}"


def verify_agent_tool_token(token: str) -> AgentToolTokenPayload:
    """校验签名 + 有效期，返回载荷。

    ⚠️ 401/403 的 message **必须统一措辞**，不区分"过期"和"签名错" ——
       给模型反馈可用于探测的信息没有好处。
       （区分放在异常**类型**上，由路由决定状态码。）
    """
    if not token or "." not in token:
        raise AgentToolTokenError()

    encoded, _, signature = token.partition(".")
    # ⚠️ 恒定时间比较 —— 防时序侧信道（与 routes/auth.py 一致）
    if not hmac.compare_digest(signature, _sign(encoded)):
        raise AgentToolTokenError()

    try:
        raw = json.loads(_b64decode(encoded))
    except (ValueError, json.JSONDecodeError) as exc:
        raise AgentToolTokenError() from exc

    if not isinstance(raw, dict):
        raise AgentToolTokenError()

    now = int(time.time())
    exp = int(raw.get("exp") or 0)
    if exp <= now:
        raise AgentToolTokenExpired()

    return AgentToolTokenPayload(
        sub=str(raw.get("sub") or ""),
        task=str(raw.get("task") or ""),
        allow=tuple(raw.get("allow") or ()),
        scope=str(raw.get("scope") or "self"),
        iat=int(raw.get("iat") or 0),
        exp=exp,
    )


def assert_tool_allowed(payload: AgentToolTokenPayload, tool_name: str) -> None:
    """工具白名单校验。"""
    if tool_name not in payload.allow:
        raise AgentToolTokenForbidden()
```

**关键点**：
- `allow` 排序去重 → 签名稳定
- TTL 钳制到 `[30, 300]`，即使调用方传 `ttl=99999` 也被压到 300
- `hmac.compare_digest` → 防时序侧信道
- 异常**类型**区分 401/403，但 `message` 统一

### 1.3 `backend/app/services/privacy.py` — 新建（PII 脱敏）

```python
"""业务数据 PII 脱敏 —— L2 侧。

⚠️ **为什么单独一个文件**：
   backend 现有的脱敏只有两个，且都**不是**业务 PII 脱敏：
     - main.py 的 _redact_sensitive_response_headers（只清响应头凭据键）
     - organization_sync 的 _sanitize_payload（只清 token/secret）
   证据正文里的手机号 / 身份证 / 邮箱没人管。

⚠️ **为什么与 agent-service 各留一份**：
   不跨服务 import（L3 不能依赖 backend，反向同理）。两侧语义必须一致 ——
   agent-service/app/grounding.py:44-60 的 redact_text 是**参照实现**。
   本文件的正则与掩码格式必须与它**逐字一致**，
   并由 backend/app/tests/test_privacy_parity.py 用同一批样例锁住一致性。

⚠️ **脱敏必须在 L2 返回前完成**：
   L2 返出的 JSON 会进 L3 内存、进模型请求体、可能进 L3 trace。
   只在 L3 脱敏的话，明文 PII 已经在网络与模型侧流过了。
"""

from __future__ import annotations

import re
from typing import Any

# ⚠️ 与 agent-service/app/grounding.py 的正则**逐字一致**。
MOBILE_RE = re.compile(r"(?<!\d)(1\d{2})\d{4}(\d{4})(?!\d)")
ID_CARD_RE = re.compile(r"(?<!\d)(\d{6})\d{8}(\d{3}[0-9Xx])(?![0-9Xx])")
EMAIL_RE = re.compile(r"([A-Za-z0-9._%+-])[A-Za-z0-9._%+-]*(@[A-Za-z0-9.-]+\.[A-Za-z]{2,})")
EMPLOYEE_NO_RE = re.compile(r"(?i)(员工编号|employeeNo|employee_no)\s*[:：=]?\s*[A-Za-z0-9_-]+")

SENSITIVE_KEYS = ("mobile", "phone", "email", "employeeno", "employee_no")


def redact_text(text: str | None) -> str:
    """脱敏文本。保留可读掩码，不记录被移除的值。

    ⚠️ 掩码格式与 agent-service/app/grounding.py::redact_text **必须一致**。
    """
    if not text:
        return ""
    value = MOBILE_RE.sub(r"\1****\2", text)
    value = ID_CARD_RE.sub(r"\1********\2", value)
    value = EMAIL_RE.sub(r"\1***\2", value)
    value = EMPLOYEE_NO_RE.sub("员工编号:***", value)
    return value


def redact_obj(obj: Any) -> Any:
    """递归脱敏对象。敏感键的值整体替换为 ***。"""
    if isinstance(obj, dict):
        return {
            key: ("***" if key.lower() in SENSITIVE_KEYS else redact_obj(value))
            for key, value in obj.items()
        }
    if isinstance(obj, list):
        return [redact_obj(item) for item in obj]
    if isinstance(obj, str):
        return redact_text(obj)
    return obj
```

### 2.2 `agent-service/app/tools.py` — 新建

```python
"""工具注册表 —— 模型可调用的只读工具。

设计约束（docs/14 五条边界）：
  B2 只读      → register() 拒绝 read_only=False
  B3 身份绑定  → identity_params 声明禁止接收的身份参数，register() 直接拒绝
  失败不外抛   → call() 返回 (data, error)，错误交给模型处理
"""

from __future__ import annotations

import json
from collections.abc import Callable
from dataclasses import dataclass, field
from typing import Any

import httpx


@dataclass(frozen=True)
class ToolSpec:
    """一个可被模型调用的只读工具。"""

    name: str
    description: str
    parameters: dict[str, Any]          # JSON Schema
    handler: Callable[[dict[str, Any]], dict[str, Any]]
    read_only: bool = True
    timeout_s: float = 8.0
    max_chars: int = 8000
    # ⚠️ 本工具**禁止**接收的身份类参数（如 employee_id）。
    #    register() 会校验 schema 里没出现这些键 —— 把 B3 从约定变成代码保证。
    identity_params: frozenset[str] = field(default_factory=frozenset)
    # 列表类工具的条目上限 —— 防止一个部门 5000 人把结果撑爆
    max_result_items: int = 200


class ToolRegistry:
    """工具注册表 —— 白名单过滤、调用分发、结果截断。"""

    def __init__(self) -> None:
        self._tools: dict[str, ToolSpec] = {}

    def register(self, spec: ToolSpec) -> None:
        """注册只读工具。写工具、身份参数冲突、schema 不合法一律拒绝。"""
        if not spec.read_only:
            raise ValueError("绩效域不允许写工具")
        if not spec.name or not spec.description:
            raise ValueError("工具必须有名称与描述")
        if spec.name in self._tools:
            raise ValueError(f"工具 {spec.name} 重复注册")

        properties = spec.parameters.get("properties")
        if properties is not None and not isinstance(properties, dict):
            raise ValueError(f"工具 {spec.name} 的 parameters.properties 必须是对象")
        if spec.identity_params & set(properties or {}):
            raise ValueError(f"工具 {spec.name} 不允许接受身份参数")

        self._tools[spec.name] = spec

    def get(self, name: str) -> ToolSpec | None:
        return self._tools.get(name)

    def schema(self, allow: set[str]) -> list[dict[str, Any]]:
        """转成模型可见的 function 定义，只暴露白名单内的工具。"""
        return [
            {
                "type": "function",
                "function": {
                    "name": t.name,
                    "description": t.description,
                    "parameters": t.parameters,
                },
            }
            for name, t in self._tools.items()
            if name in allow
        ]

    def call(
        self, name: str, args: dict[str, Any], allow: set[str]
    ) -> tuple[dict[str, Any] | None, str | None]:
        """执行工具。返回 (数据, 错误)。**错误不是异常**，交给模型处理。"""
        spec = self._tools.get(name)
        if spec is None:
            return None, f"未知工具 {name}"
        # ⚠️ 双保险：schema() 已过滤，call() 仍要独立校验
        if name not in allow:
            return None, f"该智能体无权调用工具 {name}"

        schema_err = _validate_args(spec, args)
        if schema_err:
            return None, schema_err

        try:
            data = spec.handler(args)
        except Exception as exc:  # noqa: BLE001
            # ⚠️ 异常不外抛 —— 转成错误 observation 回喂模型
            return None, f"工具 {name} 执行失败：{type(exc).__name__}"

        if not isinstance(data, dict):
            return None, f"工具 {name} 返回了非对象结果"

        data = _limit_items(data, spec.max_result_items)
        text = json.dumps(data, ensure_ascii=False)
        if len(text) > spec.max_chars:
            # ⚠️ 截断后的 preview 是**半截 JSON**，模型解析会失败。
            #    所以明确标注 truncated，并提示模型缩小范围。
            data = {
                "truncated": True,
                "truncated_at": len(text),
                "note": "结果过长已截断，请缩小查询范围或分批获取",
                "preview": text[: spec.max_chars // 2],
            }
        return data, None


def _limit_items(data: dict[str, Any], limit: int) -> dict[str, Any]:
    """对结果里的列表字段做条目上限。"""
    out = dict(data)
    for key, value in list(out.items()):
        if isinstance(value, list) and len(value) > limit:
            out[key] = value[:limit]
            out[f"{key}_truncated"] = True
            out[f"{key}_total"] = len(value)
    return out


def _validate_args(spec: ToolSpec, args: dict[str, Any]) -> str | None:
    """轻量 JSON Schema 校验。

    ⚠️ **不引入 jsonschema 依赖** —— agent-service 的 requirements 要保持最小。
       这里只校验三件最要紧的事：required 齐不齐、有没有未知键、类型对不对。
    """
    if not isinstance(args, dict):
        return f"工具 {spec.name} 的参数必须是对象"

    schema = spec.parameters or {}
    properties: dict[str, Any] = schema.get("properties") or {}

    for key in schema.get("required") or []:
        if key not in args:
            return f"工具 {spec.name} 缺少必填参数 {key}"

    unknown = set(args) - set(properties)
    if unknown:
        return f"工具 {spec.name} 不接受参数 {sorted(unknown)}"

    for key, value in args.items():
        expected = (properties.get(key) or {}).get("type")
        if expected == "string" and not isinstance(value, str):
            return f"工具 {spec.name} 的参数 {key} 必须是字符串"
        if expected == "integer" and not isinstance(value, int):
            return f"工具 {spec.name} 的参数 {key} 必须是整数"
        if expected == "boolean" and not isinstance(value, bool):
            return f"工具 {spec.name} 的参数 {key} 必须是布尔值"
    return None


# ---------------------------------------------------------------------------
# HTTP handler —— 标准实现：调 L2 的只读网关
# ---------------------------------------------------------------------------


def http_handler(base_url: str, token: str, name: str) -> Callable[[dict[str, Any]], dict[str, Any]]:
    """工具的标准实现：调 L2 的只读网关。

    ⚠️ **base_url 必须来自服务端配置**，不能来自请求体 ——
       否则攻击者能让本服务去调任意地址（SSRF）。
    """

    def handler(args: dict[str, Any]) -> dict[str, Any]:
        with httpx.Client(timeout=8.0) as client:
            response = client.post(
                f"{base_url.rstrip('/')}/v1/agent-tools/{name}",
                json=args,
                headers={"Authorization": f"Bearer {token}"},
            )
            response.raise_for_status()
            body = response.json()
            if body.get("code") != 0:
                raise RuntimeError(body.get("message") or "工具返回错误")
            return body.get("data") or {}

    return handler
```

```python
                None,

        # 解析 user 消息里的测试参数
        args: dict[str, Any] = {}
        for message in messages:
            if message.get("role") != "user":
                continue
            try:
                payload = json.loads(message.get("content") or "{}")
            except json.JSONDecodeError:
                continue
            if isinstance(payload.get("e2e_tool_args"), dict):

            break

        function = (tools[0].get("function") or {})
        return (
```

### 2.4 `agent-service/app/tool_agent.py` — 新建

```python
"""带工具调用能力的智能体基类。

与 BaseAgent 的关系：
    BaseAgent.run()      —— 单次调用 → Schema 校验 → 重试 → 降级
    ToolUsingAgent       —— 多步循环，每步可调只读工具

设计稿的 5 处修正（都是"设计稿没实现但必须实现"的）：
  1. 总超时 30 秒真的实现（设计稿 §6.2 列了但代码里没有）
  2. 工具结果**累计**字符上限（设计稿只有单次 8000，8 次 = 64000 会撑爆上下文）
  3. 连续工具失败上限（设计稿说"错误回喂模型"，但没限制能失败几次）
  4. resp.content 类型检查（可能是 str 不是 dict）
  5. tool_call 结构校验（模型可能不给 id）
"""

from __future__ import annotations

import json
import logging
import time
from typing import Any

from pydantic import ValidationError

from .base import AgentResult, BaseAgent, hash_input, redact
from .tools import ToolRegistry

logger = logging.getLogger(__name__)


class ToolUsingAgent(BaseAgent):
    """带工具调用能力的智能体基类。

    子类只需声明 allowed_tools / max_steps / max_tool_calls，
    其余由基类负责。
    """

    allowed_tools: set[str] = set()
    max_steps: int = 5
    max_tool_calls: int = 8
    # 单次请求的整体超时（秒）—— 与 review_service.AgentClient 的 30 秒对齐
    total_timeout_s: float = 30.0
    # 工具结果累计字符上限 —— 超出后不再喂新结果
    max_total_tool_chars: int = 20000
    # 连续工具失败次数上限 —— 防止模型一直重试同一个错参数
    max_tool_errors: int = 3

    def run_with_tools(self, registry: ToolRegistry, **kwargs: Any) -> AgentResult:
        started = time.perf_counter()
        raw_payload = self.build_payload(**kwargs)
        red = redact(raw_payload)
        payload = red.output
        payload["agent"] = self.name
        input_hash = hash_input(payload)

        messages: list[dict[str, Any]] = [
            {"role": "system", "content": self.system_prompt()},
            {"role": "user", "content": json.dumps(payload, ensure_ascii=False)},
        ]
        tools = registry.schema(self.allowed_tools)
        used = 0
        consecutive_errors = 0
        total_chars = 0
        last_error: str | None = None
        tool_calls_meta: list[dict[str, Any]] = []

        for step in range(self.max_steps + 1):
            # 1. 总超时
            if time.perf_counter() - started > self.total_timeout_s:
                last_error = "工具循环总超时"
                break

            resp, err = self.client.complete_with_tools(messages, tools)
            if err or resp is None:
                last_error = err or "模型无响应"
                break

            # 2. 有工具调用 → 执行并回喂
            if resp.tool_calls:
                if used + len(resp.tool_calls) > self.max_tool_calls:
                    last_error = "超出工具调用次数上限"
                    break

                messages.append(
                    {
                        "role": "assistant",
                        "content": resp.content,
                        "tool_calls": resp.tool_calls,
                    }
                )

                for call in resp.tool_calls:
                    used += 1
                    call_started = time.perf_counter()
                    name = str(call.get("name") or "")
                    # ⚠️ arguments 必须是 dict —— ModelClient 实现已负责解析，
                    #    这里再兜一层，防止某个实现忘了。
                    raw_args = call.get("arguments")
                    args = raw_args if isinstance(raw_args, dict) else {}

                    data, call_err = registry.call(name, args, self.allowed_tools)
                    latency_ms = int((time.perf_counter() - call_started) * 1000)

                    if call_err:
                        consecutive_errors += 1
                    else:
                        consecutive_errors = 0

                    content = json.dumps(
                        {"ok": call_err is None, "data": data, "error": call_err},
                        ensure_ascii=False,
                    )
                    total_chars += len(content)
                    if total_chars > self.max_total_tool_chars:
                        # ⚠️ 累计超限 —— 不再喂具体结果，让模型基于已有信息作答
                        content = json.dumps(
                            {
                                "ok": False,
                                "error": "工具结果累计超出上限，请基于已有信息作答",
                            },
                            ensure_ascii=False,
                        )

                    messages.append(
                        {
                            "role": "tool",
                            "tool_call_id": str(call.get("id") or ""),
                            "name": name,
                            "content": content,
                        }
                    )
                    tool_calls_meta.append(
                        {"tool": name, "ok": call_err is None, "latency_ms": latency_ms, "step": step}
                    )
                    self._trace_tool_call(
                        step=step, tool=name, args=args, data=data, error=call_err,
                        latency_ms=latency_ms,
                    )

                if consecutive_errors >= self.max_tool_errors:
                    last_error = "连续工具调用失败"
                    break
                continue

            # 3. 最终答案
            content = resp.content
            if not isinstance(content, dict):
                last_error = "模型返回的内容不是对象"
                break
            try:
                validated = self.output_model.model_validate(content)
            except ValidationError as exc:
                last_error = f"输出不符合 Schema：{exc.errors()[:3]}"
                break

            latency = int((time.perf_counter() - started) * 1000)
            result = AgentResult(
                agent=self.name,
                data={**validated.model_dump(), "tool_calls": tool_calls_meta},
                degraded=False,
                model=self.client.model,
                prompt_version=self.prompt_version,
                input_hash=input_hash,
                latency_ms=latency,
                removed_categories=red.removed_categories,
            )
            self._trace(result, status="ok")
            return result

        # 4. 统一降级 —— 流程不中断
        latency = int((time.perf_counter() - started) * 1000)
        result = AgentResult(
            agent=self.name,
            data={**self.fallback(payload, last_error), "tool_calls": tool_calls_meta},
            degraded=True,
            model=self.client.model,
            prompt_version=self.prompt_version,
            input_hash=input_hash,
            latency_ms=latency,
            removed_categories=red.removed_categories,
            warnings=[last_error] if last_error else [],
        )
        self._trace(result, status="degraded")
        return result

    def _trace_tool_call(
        self,
        *,
        step: int,
        tool: str,
        args: dict[str, Any],
        data: dict[str, Any] | None,
        error: str | None,
        latency_ms: int,
    ) -> None:
        """工具调用留痕。

        ⚠️ tracer 接口与 BaseAgent._trace 不同（这是**子事件**，不是一次模型调用），
           所以单独一个方法。写审计的最终责任在 L2（ai_tool_call 表），
           这里只做 L3 侧的日志。
        """
        logger.info(
            "工具调用 agent=%s step=%d tool=%s ok=%s latency_ms=%d",
            self.name, step, tool, error is None, latency_ms,
        )
```

```python
"""问答数据范围推导。

⚠️ **模型和请求体都不参与范围判定** —— 判定依据只有 access_control。
"""

from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from sqlalchemy.orm import Session

from ..domain.models import Employee
from ..exceptions import PermissionDenied
from . import access_control
from .access_control import PermissionAction

# 每个范围允许使用的工具（最小权限）
_SCOPE_TOOLS: dict[str, tuple[str, ...]] = {
    "self": ("get_metric_values", "get_evidence"),
    "direct_reports": ("get_metric_values", "get_evidence"),
    "department": ("get_metric_values", "get_evidence", "get_department_stats"),
}


@dataclass(frozen=True)
class QaScope:
    """问答范围快照。"""

    scope: str
    allowed_tools: tuple[str, ...]
    employee_ids: tuple[str, ...]
    department_ids: tuple[str, ...]


def resolve_scope(
    current_user: Employee,
    *,
    task_id: str,
    target_employee_id: str | None,
    db: Session,
) -> QaScope:
    """从已认证身份推导数据范围。

    ⚠️ `target_employee_id` 只是**候选**，必须过 access_control 才作数。
    """
    ctx = access_control.build_access_context(db, current_user)

    # 1. HR → 全员
    if ctx.has_permission(PermissionAction.VIEW_ALL_SCORES):
        employee_ids = (
            (target_employee_id,) if target_employee_id else tuple(ctx.direct_report_ids)
        )
        return QaScope(
            scope="department",
            allowed_tools=_SCOPE_TOOLS["department"],
            employee_ids=employee_ids,
            department_ids=tuple(ctx.managed_department_ids),
        )

    # 2. 指定目标 → 必须在可访问范围内
    if target_employee_id and target_employee_id != current_user.id:
        if not ctx.can_view_score_of(target_employee_id):
            raise PermissionDenied(
                current_user.id, "qa_ask", f"employee:{target_employee_id}"
            )
        return QaScope(
            scope="direct_reports",
            allowed_tools=_SCOPE_TOOLS["direct_reports"],
            employee_ids=(target_employee_id,),
            department_ids=(),
        )

    # 3. 部门负责人 → 管理范围内的部门聚合
    if ctx.managed_department_ids:
        return QaScope(
            scope="department",
            allowed_tools=_SCOPE_TOOLS["department"],
            employee_ids=tuple(ctx.direct_report_ids),
            department_ids=tuple(ctx.managed_department_ids),
        )

    # 4. 主管 → 直属下属
    if ctx.direct_report_ids:
        return QaScope(
            scope="direct_reports",
            allowed_tools=_SCOPE_TOOLS["direct_reports"],
            employee_ids=tuple(ctx.direct_report_ids),
            department_ids=(),
        )

    # 5. 普通员工 → 只有自己
    return QaScope(
        scope="self",
        allowed_tools=_SCOPE_TOOLS["self"],
        employee_ids=(current_user.id,),
        department_ids=(),
    )
```

```python
        raw, err = self._chat(messages, tools=None)
        if err:
            return None, err
        content = raw.get("content")
        if not isinstance(content, str):
            return None, "模型未返回文本"
        try:
            return json.loads(content), None
        except json.JSONDecodeError:
            return None, "模型返回的不是合法 JSON"

    def complete_with_tools(
        self,
        messages: list[dict[str, Any]],
        tools: list[dict[str, Any]],
    ) -> tuple[Completion | None, str | None]:
        """带工具的补全。"""
        raw, err = self._chat(messages, tools=tools)
        if err:
            return None, err

        parsed_calls = []
        for call in raw.get("tool_calls") or []:
            function = call.get("function") or {}
            args_raw = function.get("arguments") or "{}"
            # ⚠️ arguments 是 **JSON 字符串** —— 在这里解析掉
            try:
                arguments = json.loads(args_raw) if isinstance(args_raw, str) else dict(args_raw)
            except json.JSONDecodeError:
                arguments = {}
            parsed_calls.append(
                {
                    "id": str(call.get("id") or ""),
                    "name": str(function.get("name") or ""),
                    "arguments": arguments if isinstance(arguments, dict) else {},
                }
            )

        content_raw = raw.get("content")
        content = None
        if isinstance(content_raw, str) and content_raw.strip():
            try:
                parsed = json.loads(content_raw)
                content = parsed if isinstance(parsed, dict) else None
            except json.JSONDecodeError:
                content = None

        return Completion(content=content, tool_calls=parsed_calls), None

    def _chat(
        self, messages: list[dict[str, Any]], tools: list[dict[str, Any]] | None
    ) -> tuple[dict[str, Any], str | None]:
        headers = {"Content-Type": "application/json"}
        if self.api_key:
            headers["Authorization"] = f"Bearer {self.api_key}"
        body: dict[str, Any] = {"model": self.model, "messages": messages}
        if tools:
            body["tools"] = tools
            body["tool_choice"] = "auto"
        try:
            with httpx.Client(timeout=25.0) as client:
                response = client.post(
                    f"{self.base_url}/chat/completions", json=body, headers=headers
                )
                response.raise_for_status()
                payload = response.json()
        except Exception as exc:  # noqa: BLE001
            return {}, f"模型调用失败：{type(exc).__name__}"

        choices = payload.get("choices") or []
        if not choices:
            return {}, "模型未返回 choices"
        return choices[0].get("message") or {}, None
```
